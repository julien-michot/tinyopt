set(tinyopt_c_test_targets)

function(add_c_api_test test_name source)
	add_executable(${test_name} ${source})
	target_link_libraries(${test_name} PRIVATE tinyopt_c)
	set_target_properties(${test_name} PROPERTIES LINKER_LANGUAGE CXX)
	add_test(NAME ${test_name} COMMAND ${test_name})
	set_property(TEST ${test_name} PROPERTY LABELS Tinyopt Tests)
	list(APPEND tinyopt_c_test_targets ${test_name})
	set(tinyopt_c_test_targets ${tinyopt_c_test_targets} PARENT_SCOPE)
endfunction()

if(TINYOPT_C_API_DYNAMIC)
	add_c_api_test(tinyopt_test_c_api test_api.c)
	if(TINYOPT_C_API_FLOAT)
		add_c_api_test(tinyopt_test_c_api_float test_api_float.c)
	endif()
	add_c_api_test(tinyopt_test_c_api_block_tridiag test_api_block_tridiag.c)
	add_c_api_test(tinyopt_test_c_api_block_sparse test_api_block_sparse.c)
	if(TINYOPT_C_API_FLOAT)
		add_c_api_test(tinyopt_test_c_api_block_tridiag_float test_api_block_tridiag_float.c)
		add_c_api_test(tinyopt_test_c_api_block_sparse_float test_api_block_sparse_float.c)
	endif()
	if(TINYOPT_ENABLE_SUITESPARSE)
		add_c_api_test(tinyopt_test_c_api_sparse test_api_sparse.c)
		if(TINYOPT_C_API_FLOAT)
			add_c_api_test(tinyopt_test_c_api_sparse_float test_api_sparse_float.c)
		endif()
	endif()
else()
	add_c_api_test(tinyopt_test_c_api_no_dynamic test_api_no_dynamic.c)
endif()

foreach(dimension IN LISTS TINYOPT_C_FIXED_SIZES)
	set(fixed_suffixes d)
	if(TINYOPT_C_API_FLOAT)
		list(APPEND fixed_suffixes f)
	endif()
	foreach(suffix IN LISTS fixed_suffixes)
		set(test_name tinyopt_test_c_api_fixed_${dimension}_${suffix})
		set(test_source
			"${TINYOPT_C_FIXED_TEST_DIR}/c_api_fixed_test_${dimension}_${suffix}.c")
		add_c_api_test(${test_name} "${test_source}")
	endforeach()
endforeach()

add_custom_target(tinyopt_c_tests DEPENDS ${tinyopt_c_test_targets})
