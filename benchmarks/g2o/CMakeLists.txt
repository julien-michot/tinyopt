add_executable(tinyopt_bench_g2o bundle_adjustment.cpp)
add_benchmark_target(tinyopt_bench_g2o)

set(G2O_BENCHMARK_LIBRARIES)
foreach(component IN ITEMS core stuff types_sba solver_eigen)
  foreach(candidate IN ITEMS "g2o::${component}" "g2o_${component}")
    if (TARGET ${candidate})
      list(APPEND G2O_BENCHMARK_LIBRARIES ${candidate})
      break()
    endif()
  endforeach()
endforeach()
if (G2O_BENCHMARK_LIBRARIES)
  target_link_libraries(tinyopt_bench_g2o PRIVATE ${G2O_BENCHMARK_LIBRARIES})
else()
  target_include_directories(tinyopt_bench_g2o PRIVATE ${G2O_INCLUDE_DIRS})
  target_link_libraries(tinyopt_bench_g2o PRIVATE ${G2O_LIBRARIES})
endif()
